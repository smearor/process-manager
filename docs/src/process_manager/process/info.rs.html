6/// A lightweight snapshot of a managed process.
7///
8/// Contains metadata only — no `Child` handle. Returned by inspection methods
9/// like `get_by_label()`. Cannot be used for `is_running()`, `wait()`, or
10/// signal operations; use `stop()` / `stop_label()` for process control.
11#[derive(Debug, Clone, PartialEq, Eq)]
12#[must_use]
13pub struct ProcessInfo {
14    /// Unique identifier assigned by the `ProcessManager`.
15    pub id: ProcessId,
16
17    /// The PID of the child process.
18    pub pid: u32,
19
20    /// The program name (for error reporting).
21    pub program_name: String,
22
23    /// The label under which this process was started (for grouped operations).
24    pub label: String,
25
26    /// Whether to terminate this process when the `ProcessManager` is dropped.
27    pub terminate_on_exit: bool,
28
29    /// The configuration this process was started with.
30    pub config: Arc<ProcessConfig>,
31}
32
33impl From<&Process> for ProcessInfo {
34    fn from(process: &Process) -> Self {
35        Self {
36            id: process.id,
37            pid: process.pid,
38            program_name: process.program_name.clone(),
39            label: process.label.clone(),
40            terminate_on_exit: process.terminate_on_exit,
41            config: Arc::clone(&process.config),
42        }
43    }
44}
