28/// Polls all tracked processes via non-blocking `try_wait()` every
29/// `poll_interval`. When a process has exited, it handles automatic restart
30/// with backoff and rate limiting, or removes it from the `DashMap` and emits
31/// a `ProcessExitEvent`.
32///
33/// The reaper does not block. Backoff waits are handled by checking
34/// `next_eligible_restart` on each poll cycle.
35pub(crate) fn reaper_loop(
36    processes: Arc<DashMap<ProcessId, Process>>,
37    poll_interval: Duration,
38    exit_sender: &Sender<ProcessExitEvent>,
39    stop_flag: &AtomicBool,
40) {
41    debug!("Reaper thread started (poll_interval={:?})", poll_interval);
42    while !stop_flag.load(Ordering::Relaxed) {
43        thread::sleep(poll_interval);
44        let now = Instant::now();
45
46        // Phase 1: Detect exits and handle restart decisions
47        let mut to_remove: Vec<ProcessId> = Vec::new();
48        let mut exited: Vec<ExitedProcess> = Vec::new();
49        // Collect (crashed_id, label, spawn_sequence, strategy) for cascade processing
50        let mut cascade_origins: Vec<(ProcessId, Label, u64, SupervisorStrategy)> = Vec::new();
51
52        for mut entry in processes.iter_mut() {
53            let process = entry.value_mut();
54
55            // Skip processes in Restarting state - they have no child to poll
56            if process.state == ProcessState::Restarting {
57                continue;
58            }
59
60            // Check for stable uptime reset on Running processes
61            if process.state == ProcessState::Running
62                && let Some(restart_state) = &mut process.restart_state
63                && let Some(backoff_config) = get_backoff_config(&process.config)
64            {
65                restart_state.check_stable_uptime(backoff_config, now);
66            }
67
68            if let Some(exit_status) = process.try_wait_exit() {
69                // If cascade_flag is set, this process was stopped as part of
70                // a supervisor cascade. Emit Stopped event. If restart_on_exit
71                // is true, schedule restart (the cascade origin triggered this
72                // stop, but the process should still be restarted as part of
73                // the group restart).
74                if process.cascade_flag {
75                    let should_restart_cascade = process.config.restart_on_exit;
76                    let restart_config = if should_restart_cascade {
77                        Some((Arc::clone(&process.config), process.label.clone()))
78                    } else {
79                        None
80                    };
81                    exited.push(ExitedProcess {
82                        id: process.id,
83                        label: process.label.clone(),
84                        pid: process.pid,
85                        restart_on_exit: process.config.restart_on_exit,
86                        exit_status: Some(exit_status),
87                        state: ProcessState::Stopped,
88                        restart_config,
89                    });
90                    // Don't remove if it will restart
91                    if !should_restart_cascade {
92                        to_remove.push(process.id);
93                    }
94                    // Reset cascade_flag - the cascade is complete for this process
95                    process.cascade_flag = false;
96                    continue;
97                }
98
99                let base_state = if exit_status.success() {
100                    ProcessState::Stopped
101                } else {
102                    ProcessState::Crashed
103                };
104
105                // Determine if restart should be triggered
106                let restart_triggered = process.config.restart_on_exit && should_restart(&process.config.restart_trigger, base_state);
107
108                // Determine final event state (rate-limit check before emission)
109                let event_state = if restart_triggered {
110                    if let Some(restart_state) = &process.restart_state {
111                        if let Some(backoff_config) = get_backoff_config(&process.config) {
112                            if restart_state.is_rate_limited(backoff_config) {
113                                ProcessState::Failed
114                            } else {
115                                base_state
116                            }
117                        } else {
118                            // Immediate policy - no rate limiting
119                            base_state
120                        }
121                    } else {
122                        base_state
123                    }
124                } else {
125                    base_state
126                };
127
128                let restart_config = if restart_triggered && event_state != ProcessState::Failed {
129                    Some((Arc::clone(&process.config), process.label.clone()))
130                } else {
131                    None
132                };
133
134                // If this is a crash (not clean exit) and restart is triggered,
135                // record the cascade origin for Phase 1b
136                if restart_triggered && event_state != ProcessState::Failed && process.config.supervisor_strategy != SupervisorStrategy::OneForOne {
137                    cascade_origins.push((process.id, process.label.clone(), process.spawn_sequence, process.config.supervisor_strategy));
138                }
139
140                exited.push(ExitedProcess {
141                    id: process.id,
142                    label: process.label.clone(),
143                    pid: process.pid,
144                    restart_on_exit: process.config.restart_on_exit,
145                    exit_status: Some(exit_status),
146                    state: event_state,
147                    restart_config,
148                });
149
150                // Mark for removal only if not restarting
151                if event_state == ProcessState::Failed || !restart_triggered {
152                    to_remove.push(process.id);
153                }
154            } else if process.state == ProcessState::Starting {
155                // Process is alive and was in Starting state - transition to Running
156                process.state = ProcessState::Running;
157            }
158        }
159
160        // Phase 1b: Handle supervisor strategy cascades
161        // For each crashed process with OneForAll or RestForOne, flag and
162        // send kill signals to affected group members.
163        let mut cascade_targets: Vec<ProcessId> = Vec::new();
164        for (origin_id, origin_label, origin_seq, strategy) in &cascade_origins {
165            let mut targets: Vec<ProcessId> = Vec::new();
166            for entry in processes.iter() {
167                let process = entry.value();
168                // Skip the origin process itself
169                if process.id == *origin_id {
170                    continue;
171                }
172                // Only target processes in the same label group
173                if process.label != *origin_label {
174                    continue;
175                }
176                // Only target Running or Starting processes
177                if process.state != ProcessState::Running && process.state != ProcessState::Starting {
178                    continue;
179                }
180                match strategy {
181                    SupervisorStrategy::OneForAll => {
182                        targets.push(*entry.key());
183                    }
184                    SupervisorStrategy::RestForOne => {
185                        // Only stop processes started after the crashed one
186                        if process.spawn_sequence > *origin_seq {
187                            targets.push(*entry.key());
188                        }
189                    }
190                    SupervisorStrategy::OneForOne => {}
191                }
192            }
193            cascade_targets.extend(targets);
194        }
195
196        // Flag and send kill signals to cascade targets (async, non-blocking)
197        for target_id in &cascade_targets {
198            if let Some(mut entry) = processes.get_mut(target_id) {
199                let process = entry.value_mut();
200                // Safety: pid 0 would broadcast the signal to the entire
201                // process group. Waiting/Failed processes have pid 0 and
202                // should never be cascade targets, but guard defensively.
203                if process.pid == 0 {
204                    warn!("Reaper: skipping cascade kill for process {} with pid=0", target_id);
205                    continue;
206                }
207                process.cascade_flag = true;
208                process.state = ProcessState::Stopping;
209                let kill_signal = process.config.kill_signal;
210                let pid = process.pid;
211                // Send kill signal asynchronously - do not wait for exit
212                if let Err(e) = process.send_signal(kill_signal)
213                    && e.raw_os_error() != Some(libc::ESRCH)
214                {
215                    warn!("Reaper: failed to send cascade kill signal to process {} (pid={}): {}", target_id, pid, e);
216                }
217                debug!("Reaper: sent cascade kill signal to process {} (pid={})", target_id, pid);
218            }
219        }
220
221        // Phase 2: Remove non-restarting processes and emit events
222        for id in &to_remove {
223            processes.remove(id);
224        }
225
226        for exited_process in &exited {
227            let event = ProcessExitEvent {
228                id: exited_process.id,
229                label: exited_process.label.clone(),
230                pid: exited_process.pid,
231                restart_on_exit: exited_process.restart_on_exit,
232                exit_status: exited_process.exit_status,
233                state: exited_process.state,
234            };
235            debug!("Reaper: process {} (pid={}) exited (state={}), emitting event", event.id, event.pid, event.state);
236            if exit_sender.send(event).is_err() {
237                warn!("Reaper: exit event channel closed, stopping reaper thread");
238                return;
239            }
240        }
241
242        // Phase 3: Handle restart for processes that will restart
243        for exited_process in &exited {
244            if exited_process.restart_config.is_none() {
245                continue;
246            }
247
248            let (config, _label) = exited_process.restart_config.as_ref().unwrap();
249            let id = exited_process.id;
250
251            // Get the process entry (it was not removed since it's restarting)
252            let mut entry = match processes.get_mut(&id) {
253                Some(entry) => entry,
254                None => continue,
255            };
256            let process = entry.value_mut();
257
258            // Release OS resources before backoff wait
259            if let Some(child) = process.child.as_mut() {
260                let _ = child.wait();
261            }
262            process.child = None;
263            process.join_readers(Duration::from_millis(500));
264
265            // Record restart and compute backoff delay
266            if let Some(restart_state) = &mut process.restart_state {
267                restart_state.record_restart();
268                if let Some(backoff_config) = get_backoff_config(config) {
269                    let delay = restart_state.current_delay(backoff_config);
270                    restart_state.schedule_restart(now + delay);
271                    debug!(
272                        "Reaper: process {} scheduled for restart in {:?} (restart_count={})",
273                        id,
274                        delay,
275                        restart_state.restart_count()
276                    );
277                } else {
278                    // Immediate policy - eligible right away
279                    restart_state.schedule_restart(now);
280                    debug!("Reaper: process {} scheduled for immediate restart", id);
281                }
282            }
283
284            // Set Restarting state
285            process.state = ProcessState::Restarting;
286        }
287
288        // Phase 4: Check processes in Restarting state for eligible restarts
289        let mut restart_ids: Vec<ProcessId> = Vec::new();
290        for entry in processes.iter() {
291            if entry.value().state == ProcessState::Restarting
292                && let Some(restart_state) = &entry.value().restart_state
293                && restart_state.is_eligible_for_restart(now)
294            {
295                restart_ids.push(*entry.key());
296            }
297        }
298
299        for id in restart_ids {
300            perform_restart(&processes, id, exit_sender);
301        }
302
303        // Phase 5: Check Waiting processes - spawn when deps are Running,
304        // fail-fast when a dependency is terminal, timeout if deps not ready.
305        //
306        // We collect Waiting process IDs first, then process each one
307        // individually. This avoids nested DashMap iteration which can
308        // deadlock when the outer iterator holds a shard read lock and
309        // the inner iteration (for label resolution) tries to acquire
310        // the same shard.
311        let waiting_ids: Vec<ProcessId> = processes
312            .iter()
313            .filter(|e| e.value().state == ProcessState::Waiting)
314            .map(|e| *e.key())
315            .collect();
316
317        let mut waiting_to_spawn: Vec<ProcessId> = Vec::new();
318        let mut waiting_to_fail: Vec<(ProcessId, ProcessState)> = Vec::new();
319
320        for id in waiting_ids {
321            let entry = match processes.get(&id) {
322                Some(e) => e,
323                None => continue,
324            };
325            let process = entry.value();
326
327            // Check dependency timeout
328            if let Some(since) = process.waiting_since {
329                let elapsed = since.elapsed();
330                let timeout = Duration::from_millis(process.config.dependency_timeout_ms);
331                if elapsed >= timeout {
332                    debug!("Reaper: Waiting process {} dependency timeout (elapsed={:?}, timeout={:?})", id, elapsed, timeout);
333                    waiting_to_fail.push((id, ProcessState::Failed));
334                    continue;
335                }
336            }
337
338            let all_resolved = process.resolved_deps.len() == process.config.depends_on.len();
339
340            if !all_resolved {
341                // Some deps not yet resolved - try to resolve remaining labels.
342                // Collect info needed for resolution without holding the borrow.
343                let deps_to_resolve: Vec<DependencyRef> = process
344                    .config
345                    .depends_on
346                    .iter()
347                    .filter(|dep| match dep {
348                        DependencyRef::Label(label) => !process
349                            .resolved_deps
350                            .iter()
351                            .any(|rid| processes.get(rid).is_some_and(|e| e.value().label == *label)),
352                        DependencyRef::Id(id) => !process.resolved_deps.contains(id),
353                    })
354                    .cloned()
355                    .collect();
356                let existing_resolved = process.resolved_deps.clone();
357
358                drop(entry);
359
360                let mut newly_resolved: Vec<ProcessId> = Vec::new();
361                for dep in &deps_to_resolve {
362                    match dep {
363                        DependencyRef::Label(label) => {
364                            // Try to find a Running process with this label
365                            let found = processes
366                                .iter()
367                                .find(|e| e.value().label == *label && e.value().state == ProcessState::Running)
368                                .map(|e| *e.key());
369                            if let Some(dep_id) = found {
370                                newly_resolved.push(dep_id);
371                            }
372                        }
373                        DependencyRef::Id(dep_id) => {
374                            newly_resolved.push(*dep_id);
375                        }
376                    }
377                }
378
379                if !newly_resolved.is_empty() {
380                    if let Some(mut entry) = processes.get_mut(&id) {
381                        entry.value_mut().resolved_deps.extend(newly_resolved);
382                    }
383                    // Re-check if all deps are now resolved and running
384                    let process = match processes.get(&id) {
385                        Some(e) => e,
386                        None => continue,
387                    };
388                    let all_resolved = process.resolved_deps.len() == process.config.depends_on.len();
389                    let all_running = process
390                        .resolved_deps
391                        .iter()
392                        .all(|dep_id| processes.get(dep_id).is_some_and(|e| e.value().state == ProcessState::Running));
393                    if all_resolved && all_running {
394                        waiting_to_spawn.push(id);
395                    }
396                    continue;
397                }
398
399                // Check for terminal dependencies among already-resolved ones
400                let terminal_dep = existing_resolved.iter().find_map(|dep_id| match processes.get(dep_id) {
401                    Some(dep) => {
402                        if dep.state == ProcessState::Failed {
403                            return Some(*dep_id);
404                        }
405                        if dep.state == ProcessState::Stopped && !dep.config.restart_on_exit {
406                            return Some(*dep_id);
407                        }
408                        None
409                    }
410                    None => Some(*dep_id),
411                });
412
413                if terminal_dep.is_some() {
414                    waiting_to_fail.push((id, ProcessState::Failed));
415                }
416                continue;
417            }
418
419            // All deps resolved - check if they're all Running.
420            // Once resolved, we NEVER re-resolve. This ensures label binding
421            // persists: if a resolved dep is gone, we fail-fast rather than
422            // binding to a new process with the same label.
423            let all_running = process
424                .resolved_deps
425                .iter()
426                .all(|dep_id| processes.get(dep_id).is_some_and(|e| e.value().state == ProcessState::Running));
427
428            if all_running {
429                waiting_to_spawn.push(id);
430            } else {
431                // Check for terminal dependencies
432                let terminal_dep = process.resolved_deps.iter().find_map(|dep_id| match processes.get(dep_id) {
433                    Some(dep) => {
434                        if dep.state == ProcessState::Failed {
435                            return Some(*dep_id);
436                        }
437                        if dep.state == ProcessState::Stopped && !dep.config.restart_on_exit {
438                            return Some(*dep_id);
439                        }
440                        None
441                    }
442                    None => Some(*dep_id),
443                });
444
445                if terminal_dep.is_some() {
446                    waiting_to_fail.push((id, ProcessState::Failed));
447                }
448            }
449        }
450
451        // Spawn Waiting processes whose deps are all Running
452        for id in waiting_to_spawn {
453            spawn_waiting_process(&processes, id, exit_sender);
454        }
455
456        // Fail Waiting processes with terminal dependencies
457        for (id, fail_state) in waiting_to_fail {
458            if let Some(mut entry) = processes.get_mut(&id) {
459                let process = entry.value_mut();
460                process.state = fail_state;
461                let event = ProcessExitEvent {
462                    id,
463                    label: process.label.clone(),
464                    pid: process.pid,
465                    restart_on_exit: process.config.restart_on_exit,
466                    exit_status: None,
467                    state: fail_state,
468                };
469                debug!("Reaper: Waiting process {} failed (dependency terminal), emitting event", id);
470                drop(entry);
471                processes.remove(&id);
472                if exit_sender.send(event).is_err() {
473                    warn!("Reaper: exit event channel closed during Waiting fail-fast");
474                    return;
475                }
476            }
477        }
478
479        // Phase 6: Check Running processes for terminal dependencies.
480        // If a Running process has a resolved dependency that is gone (removed)
481        // or in a terminal state (Failed, Stopped without restart), fail-fast
482        // the dependent process. This ensures label binding persists: the
483        // dependent does not switch to a new process with the same label.
484        let running_with_deps: Vec<ProcessId> = processes
485            .iter()
486            .filter(|e| e.value().state == ProcessState::Running && !e.value().resolved_deps.is_empty())
487            .map(|e| *e.key())
488            .collect();
489
490        for id in running_with_deps {
491            let entry = match processes.get(&id) {
492                Some(e) => e,
493                None => continue,
494            };
495            let process = entry.value();
496            let resolved_deps = process.resolved_deps.clone();
497
498            let terminal_dep = resolved_deps.iter().find_map(|dep_id| match processes.get(dep_id) {
499                Some(dep) => {
500                    if dep.state == ProcessState::Failed {
501                        return Some(*dep_id);
502                    }
503                    if dep.state == ProcessState::Stopped && !dep.config.restart_on_exit {
504                        return Some(*dep_id);
505                    }
506                    None
507                }
508                None => Some(*dep_id),
509            });
510
511            if terminal_dep.is_some() {
512                let label = process.label.clone();
513                let pid = process.pid;
514                let restart_on_exit = process.config.restart_on_exit;
515                drop(entry);
516
517                if let Some(mut entry) = processes.get_mut(&id) {
518                    let process = entry.value_mut();
519                    process.state = ProcessState::Failed;
520                    let kill_signal = process.config.kill_signal;
521                    if pid > 0 {
522                        let _ = process.send_signal(kill_signal);
523                    }
524                    drop(entry);
525                }
526                processes.remove(&id);
527
528                let event = ProcessExitEvent {
529                    id,
530                    label,
531                    pid,
532                    restart_on_exit,
533                    exit_status: None,
534                    state: ProcessState::Failed,
535                };
536                debug!("Reaper: Running process {} failed (dependency terminal), emitting event", id);
537                if exit_sender.send(event).is_err() {
538                    warn!("Reaper: exit event channel closed during Running fail-fast");
539                    return;
540                }
541            }
542        }
543    }
544    debug!("Reaper thread stopped");
545}

547/// Check if a restart should be triggered based on the trigger config and exit state.
548fn should_restart(trigger: &RestartTrigger, state: ProcessState) -> bool {
549    match trigger {
550        RestartTrigger::CrashOnly => state == ProcessState::Crashed,
551        RestartTrigger::Always => true,
552    }
553}

555/// Get the `BackoffConfig` from the restart policy, if any.
556fn get_backoff_config(config: &crate::config::ProcessConfig) -> Option<&BackoffConfig> {
557    match &config.restart_policy {
558        RestartPolicy::Backoff(config) => Some(config),
559        RestartPolicy::Immediate => None,
560    }

563/// Attempt to restart a process that is in `Restarting` state and eligible.
564///
565/// If the process has `depends_on`, dependencies are re-resolved before
566/// spawning. If dependencies are not yet `Running`, the process transitions
567/// to `Waiting` instead of spawning. The reaper's Phase 5 will then spawn
568/// it once all dependencies are `Running`.
569fn perform_restart(processes: &DashMap<ProcessId, Process>, id: ProcessId, exit_sender: &Sender<ProcessExitEvent>) {
570    let mut entry = match processes.get_mut(&id) {
571        Some(entry) => entry,
572        None => return,
573    };
574    let process = entry.value_mut();
575
576    let config = Arc::clone(&process.config);
577    let label = process.label.clone();
578
579    // Check dependencies before restarting. If the process has depends_on
580    // and not all dependencies are Running, transition to Waiting instead
581    // of spawning. This ensures dependency-ordered restart in OneForAll
582    // and RestForOne strategies.
583    if !config.depends_on.is_empty() {
584        let resolved = resolve_dependencies(processes, &config.depends_on).unwrap_or_default();
585        let all_resolved = resolved.len() == config.depends_on.len();
586        let all_running = all_resolved && all_deps_running(processes, &resolved);
587
588        if !all_running {
589            // Dependencies not ready - transition to Waiting
590            process.state = ProcessState::Waiting;
591            process.waiting_since = Some(Instant::now());
592            process.resolved_deps = resolved;
593            process.cascade_flag = false;
594            debug!("Reaper: process {} transitioning to Waiting on restart (deps not ready)", id);
595            return;
596        }
597
598        // Deps are ready - update resolved_deps for consistency
599        process.resolved_deps = resolved;
600    }
601
602    // Spawn the new OS process
603    let spawn_result = spawn_process(&config, &label);
604
605    match spawn_result {
606        Ok(spawn) => {
607            let new_pid = spawn.child.id();
608
609            // Update the existing entry in-place
610            process.child = Some(spawn.child);
611            process.pid = new_pid;
612            process.stdout_reader = spawn.stdout_reader;
613            process.stderr_reader = spawn.stderr_reader;
614            process.state = ProcessState::Starting;
615
616            // Mark started for uptime tracking
617            if let Some(restart_state) = &mut process.restart_state {
618                restart_state.mark_started(Instant::now());
619                restart_state.cancel_pending_restart();
620            }
621
622            debug!("Reaper: process {} restarted successfully (new_pid={})", id, new_pid);
623        }
624        Err(err) => {
625            // Spawn failed - terminal
626            warn!("Reaper: failed to restart process {} (spawn error: {}), setting Failed", id, err);
627            process.state = ProcessState::Failed;
628
629            let event = ProcessExitEvent {
630                id,
631                label: process.label.clone(),
632                pid: process.pid,
633                restart_on_exit: process.config.restart_on_exit,
634                exit_status: None,
635                state: ProcessState::Failed,
636            };
637            drop(entry);
638            processes.remove(&id);
639            debug!("Reaper: process {} removed after spawn failure (state=Failed)", id);
640            if exit_sender.send(event).is_err() {
641                warn!("Reaper: exit event channel closed during spawn failure");
642            }
643        }
644    }
645}

647/// Spawn a process that was in `Waiting` state now that its dependencies are `Running`.
648///
649/// Updates the process in-place: sets the child handle, PID, state to `Starting`,
650/// and marks started for restart tracking.
651fn spawn_waiting_process(processes: &DashMap<ProcessId, Process>, id: ProcessId, exit_sender: &Sender<ProcessExitEvent>) {
652    let mut entry = match processes.get_mut(&id) {
653        Some(entry) => entry,
654        None => return,
655    };
656    let process = entry.value_mut();
657
658    let config = Arc::clone(&process.config);
659    let label = process.label.clone();
660
661    let spawn_result = spawn_process(&config, &label);
662
663    match spawn_result {
664        Ok(spawn) => {
665            let new_pid = spawn.child.id();
666            process.child = Some(spawn.child);
667            process.pid = new_pid;
668            process.stdout_reader = spawn.stdout_reader;
669            process.stderr_reader = spawn.stderr_reader;
670            process.state = ProcessState::Starting;
671            process.waiting_since = None;
672
673            if let Some(restart_state) = &mut process.restart_state {
674                restart_state.mark_started(Instant::now());
675            }
676
677            debug!("Reaper: Waiting process {} spawned (pid={}), transitioning to Starting", id, new_pid);
678        }
679        Err(err) => {
680            warn!("Reaper: failed to spawn Waiting process {} (spawn error: {}), setting Failed", id, err);
681            process.state = ProcessState::Failed;
682
683            let event = ProcessExitEvent {
684                id,
685                label: process.label.clone(),
686                pid: process.pid,
687                restart_on_exit: process.config.restart_on_exit,
688                exit_status: None,
689                state: ProcessState::Failed,
690            };
691            drop(entry);
692            processes.remove(&id);
693            debug!("Reaper: Waiting process {} removed after spawn failure (state=Failed)", id);
694            if exit_sender.send(event).is_err() {
695                warn!("Reaper: exit event channel closed during Waiting spawn failure");
696            }
697        }
698    }
699}

701/// Spawn a new OS process from the given config and label.
702fn spawn_process(config: &crate::config::ProcessConfig, label: &Label) -> std::io::Result<SpawnResult> {
703    let program_name = config.command.clone();
704
705    // 1. Resolve executable
706    let (executable, args) = if config.shell {
707        let full_command = if config.args.is_empty() {
708            config.command.clone()
709        } else {
710            format!("{} {}", config.command, config.args.join(" "))
711        };
712        ("sh".to_string(), vec!["-c".to_string(), full_command])
713    } else {
714        let exe = if PathBuf::from(&config.command).is_absolute() {
715            config.command.clone()
716        } else {
717            which::which(&config.command).map_err(std::io::Error::other)?.to_string_lossy().to_string()
718        };
719        (exe, config.args.clone())
720    };
721
722    // 2. Build Command
723    let mut command = Command::new(&executable);
724    command.args(&args);
725
726    // 3. Set environment variables
727    for (key, value) in &config.env {
728        command.env(key, value);
729    }
730    if let Some(socket) = &config.socket {
731        command.env("WAYLAND_DISPLAY", socket);
732    }
733
734    // 4. Set working directory
735    if let Some(working_dir) = &config.working_dir {
736        command.current_dir(working_dir);
737    }
738
739    // 5. Configure stdio
740    command.stdin(config.stdin.to_stdio());
741    command.stdout(config.stdout.to_stdio());
742    command.stderr(config.stderr.to_stdio());
743
744    // 6. Forked mode: setsid() via pre_exec
745    if config.forked {
746        use std::os::unix::process::CommandExt;
747        unsafe {
748            command.pre_exec(|| {
749                if libc::setsid() < 0 {
750                    return Err(std::io::Error::last_os_error());
751                }
752                Ok(())
753            });
754        }
755    }
756
757    // 7. Spawn
758    let mut child = command.spawn()?;
759
760    // 8. Spawn reader threads for piped stdout/stderr
761    let stdout_reader = if config.stdout == crate::config::StdioConfig::Piped
762        && let Some(stdout) = child.stdout.take()
763    {
764        let program_name_clone = program_name.clone();
765        let label_clone = label.clone();
766        thread::Builder::new()
767            .name(format!("stdout-reader-{}", program_name_clone))
768            .spawn(move || {
769                use std::io::BufRead;
770                let reader = std::io::BufReader::new(stdout);
771                for line in reader.lines() {
772                    match line {
773                        Ok(line) => debug!("[{}:{}] stdout: {}", label_clone, program_name_clone, line),
774                        Err(_) => break,
775                    }
776                }
777            })
778            .ok()
779    } else {
780        None
781    };
782    let stderr_reader = if config.stderr == crate::config::StdioConfig::Piped
783        && let Some(stderr) = child.stderr.take()
784    {
785        let program_name_clone = program_name.clone();
786        let label_clone = label.clone();
787        thread::Builder::new()
788            .name(format!("stderr-reader-{}", program_name_clone))
789            .spawn(move || {
790                use std::io::BufRead;
791                let reader = std::io::BufReader::new(stderr);
792                for line in reader.lines() {
793                    match line {
794                        Ok(line) => warn!("[{}:{}] stderr: {}", label_clone, program_name_clone, line),
795                        Err(_) => break,
796                    }
797                }
798            })
799            .ok()
800    } else {
801        None
802    };
803
804    Ok(SpawnResult {
805        child,
806        stdout_reader,
807        stderr_reader,
808    })
