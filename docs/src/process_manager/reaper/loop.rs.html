14/// Polls all tracked processes via non-blocking `try_wait()` every
15/// `poll_interval`. When a process has exited, it is removed from the
16/// `DashMap` and a `ProcessExitEvent` is sent to the channel.
17pub(crate) fn reaper_loop(
18    processes: Arc<DashMap<ProcessId, Process>>,
19    poll_interval: Duration,
20    exit_sender: &Sender<ProcessExitEvent>,
21    stop_flag: &AtomicBool,
22) {
23    debug!("Reaper thread started (poll_interval={:?})", poll_interval);
24    while !stop_flag.load(Ordering::Relaxed) {
25        thread::sleep(poll_interval);
26
27        // Collect exited processes with their exit status
28        let mut exited: Vec<(ProcessId, String, u32, bool, Option<std::process::ExitStatus>)> = Vec::new();
29        let mut to_remove: Vec<ProcessId> = Vec::new();
30
31        for mut entry in processes.iter_mut() {
32            let process = entry.value_mut();
33            if let Some(exit_status) = process.try_wait_exit() {
34                exited.push((process.id, process.label.clone(), process.pid, process.config.restart_on_exit, Some(exit_status)));
35                to_remove.push(process.id);
36            }
37        }
38
39        // Remove exited processes and emit events
40        for (id, label, pid, restart_on_exit, exit_status) in exited {
41            processes.remove(&id);
42            let event = ProcessExitEvent {
43                id,
44                label,
45                pid,
46                restart_on_exit,
47                exit_status,
48            };
49            debug!("Reaper: process {} (pid={}) exited, emitting event", event.id, event.pid);
50            if exit_sender.send(event).is_err() {
51                warn!("Reaper: exit event channel closed, stopping reaper thread");
52                break;
53            }
54        }
55    }
56    debug!("Reaper thread stopped");
57}
