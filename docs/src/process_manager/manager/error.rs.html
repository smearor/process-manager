5/// Error returned by `ProcessManager` operations.
6#[derive(Debug, Error)]
7#[must_use]
8pub enum ProcessManagerError {
9    /// No process with the given ID was found in the manager.
10    #[error("Process with ID {0} not found")]
11    NotFound(ProcessId),
12    /// The executable was not found in `PATH` and is not an absolute path.
13    #[error("Executable '{0}' not found in PATH")]
14    ExecutableNotFound(String),
15    /// Failed to spawn the child process.
16    #[error("Failed to spawn process: {0}")]
17    SpawnFailed(#[from] std::io::Error),
18    /// Failed to send a signal to the process.
19    #[error("Failed to send signal to process {0}: {1}")]
20    SignalFailed(u32, String),
21    /// A ProcessConfig validation error.
22    #[error("Process config error: {0}")]
23    ConfigError(#[from] ProcessConfigError),
24    /// Failed to spawn the reaper thread.
25    #[error("Failed to spawn reaper thread: {0}")]
26    ReaperThreadFailed(std::io::Error),
27    /// One or more processes failed to stop during a batch operation.
28    #[error("{0}")]
29    StopMany(#[from] StopManyError),
30}
31
32/// Aggregated errors from stopping multiple processes.
33///
34/// Returned by `stop_many` when one or more processes could not be stopped.
35/// Each individual error is preserved so the caller can inspect all failures.
36#[derive(Debug, Error)]
37#[error("{count} error(s) stopping processes: {errors}", count = self.errors.len(), errors = self.errors.iter().map(|e| e.to_string()).collect::<Vec<_>>().join(", "))]
38#[must_use]
39pub struct StopManyError {
40    /// All individual errors encountered during the batch stop.
41    pub errors: Vec<ProcessManagerError>,
42}
43
44impl StopManyError {
45    /// Create a `StopManyError` from a non-empty vector of errors.
46    pub fn new(errors: Vec<ProcessManagerError>) -> Self {
47        Self { errors }
48    }
49
50    /// Whether any errors were collected.
51    pub fn is_empty(&self) -> bool {
52        self.errors.is_empty()
53    }
54}
